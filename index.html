from fastapi import FastAPI from fastapi.middleware.cors import CORSMiddleware import requests from datetime import datetime import yfinance as yf app = FastAPI() # ========================================= # CONFIG # ========================================= SYMBOL = "CONNPLEX" DISPLAY_NAME = "CONNPLEX CINEMAS LTD - SM.NS" # ========================================= # CORS # ========================================= app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # ========================================= # SESSION # ========================================= session = requests.Session() HEADERS = { "User-Agent": ( "Mozilla/5.0 (Windows NT 10.0; Win64; x64) " "AppleWebKit/537.36 (KHTML, like Gecko) " "Chrome/122.0.0.0 Safari/537.36" ), "Accept": "application/json,text/plain,*/*", "Accept-Language": "en-US,en;q=0.9", "Referer": "https://www.google.com/", "Connection": "keep-alive" } # ========================================= # ROOT # ========================================= @app.get("/") def root(): return {"status": "API running"} # ========================================= # NSE FETCH # ========================================= def fetch_stock_nse(symbol): try: # Initialize NSE session session.get( "https://www.nseindia.com", headers=HEADERS, timeout=10 ) url = f"https://www.nseindia.com/api/quote-equity?symbol={symbol}" response = session.get( url, headers=HEADERS, timeout=10 ) print("NSE STATUS:", response.status_code) if response.status_code != 200: print("NSE FAILED:", response.text[:300]) return None if not response.text.strip(): print("NSE EMPTY RESPONSE") return None data = response.json() if "priceInfo" not in data: print("NSE INVALID DATA") return None price_data = data["priceInfo"] high = price_data.get("intraDayHighLow", {}).get("max") low = price_data.get("intraDayHighLow", {}).get("min") price = price_data.get("lastPrice") vwap = data.get("securityWiseDP", {}).get("vwap") # Manual VWAP fallback if not vwap and high and low and price: vwap = round((high + low + price) / 3, 2) return { "symbol": DISPLAY_NAME, "price": price or 0, "open": price_data.get("open") or 0, "high": high or 0, "low": low or 0, "prev_close": price_data.get("previousClose") or 0, "change": price_data.get("change") or 0, "change_percent": round( float(price_data.get("pChange", 0)), 2 ), "vwap": vwap or 0, } except Exception as e: print("NSE ERROR:", str(e)) return None # ========================================= # YAHOO FINANCE FALLBACK # ========================================= def fetch_stock_yfinance(symbol): try: ticker = yf.Ticker(symbol + ".NS") # Use history instead of .info hist = ticker.history(period="2d") if hist.empty: print("YFINANCE EMPTY") return None latest = hist.iloc[-1] price = round(float(latest["Close"]), 2) open_price = round(float(latest["Open"]), 2) high = round(float(latest["High"]), 2) low = round(float(latest["Low"]), 2) prev_close = price change = 0 change_percent = 0 if len(hist) >= 2: prev_close = round(float(hist.iloc[-2]["Close"]), 2) change = round(price - prev_close, 2) if prev_close != 0: change_percent = round( (change / prev_close) * 100, 2 ) vwap = round((high + low + price) / 3, 2) return { "symbol": DISPLAY_NAME, "price": price, "open": open_price, "high": high, "low": low, "prev_close": prev_close, "change": change, "change_percent": change_percent, "vwap": vwap, } except Exception as e: print("YFINANCE ERROR:", str(e)) return None # ========================================= # STOCK WRAPPER # ========================================= def fetch_stock(symbol): # Try NSE first data = fetch_stock_nse(symbol) if data: print("NSE SUCCESS") return data # Fallback to Yahoo data = fetch_stock_yfinance(symbol) if data: print("YFINANCE SUCCESS") return data # Final safe fallback return { "symbol": DISPLAY_NAME, "price": 0, "open": 0, "high": 0, "low": 0, "prev_close": 0, "change": 0, "change_percent": 0, "vwap": 0, "error": "Data temporarily unavailable" } # ========================================= # INDEX DATA # ========================================= def fetch_index_data(): try: nifty = None sensex = None try: nifty_hist = yf.Ticker("^NSEI").history(period="1d") if not nifty_hist.empty: nifty = round( float(nifty_hist.iloc[-1]["Close"]), 2 ) except Exception as e: print("NIFTY ERROR:", str(e)) try: sensex_hist = yf.Ticker("^BSESN").history(period="1d") if not sensex_hist.empty: sensex = round( float(sensex_hist.iloc[-1]["Close"]), 2 ) except Exception as e: print("SENSEX ERROR:", str(e)) return { "nifty": nifty, "sensex": sensex } except Exception as e: print("INDEX ERROR:", str(e)) return { "nifty": None, "sensex": None } # ========================================= # MAIN API # ========================================= @app.get("/data") def get_stock_data(): stock_data = fetch_stock(SYMBOL) index_data = fetch_index_data() return { "stock": stock_data, "index": index_data, "timestamp": datetime.now().strftime( "%d-%b-%Y %H:%M:%S IST" ) }
